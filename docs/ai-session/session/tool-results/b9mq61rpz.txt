     1	"""Scripted models built on Pydantic AI's FunctionModel: deterministic, offline, inspectable."""
     2	
     3	from __future__ import annotations
     4	
     5	from collections.abc import Callable
     6	
     7	from pydantic_ai import ModelMessage, ModelResponse, TextPart, ToolCallPart, ToolReturnPart
     8	from pydantic_ai.models.function import AgentInfo, FunctionModel
     9	from pydantic_ai.usage import RequestUsage
    10	
    11	USAGE = RequestUsage(input_tokens=100, output_tokens=20)
    12	
    13	
    14	def final(payload: dict, *, model_name: str = "fake") -> ModelResponse:
    15	    return ModelResponse(parts=[ToolCallPart("final_result", payload)], usage=USAGE, model_name=model_name)
    16	
    17	
    18	def answer_model(
    19	    text: str = "ok", *, status: str = "answered", sources: list[str] | None = None, tools: list[str] | None = None,
    20	    model_name: str = "fake-answer",
    21	) -> FunctionModel:
    22	    """Calls each tool in `tools` once (one per request), then returns the structured answer."""
    23	
    24	    def fn(messages: list[ModelMessage], info: AgentInfo) -> ModelResponse:
    25	        called = {p.tool_name for m in messages if isinstance(m, ModelResponse) for p in m.parts if isinstance(p, ToolCallPart)}
    26	        for t in tools or []:
    27	            if t not in called:
    28	                return ModelResponse(parts=[ToolCallPart(t, {})], usage=USAGE, model_name=model_name)
    29	        return final({"status": status, "answer": text, "sources": sources or [], "limitations": []}, model_name=model_name)
    30	
    31	    return FunctionModel(fn, model_name=model_name)
    32	
    33	
    34	def screening_model(*, in_scope: bool = True, injection: bool = False, model_name: str = "fake-guard") -> FunctionModel:
    35	    def fn(messages: list[ModelMessage], info: AgentInfo) -> ModelResponse:
    36	        return final({"in_scope": in_scope, "injection": injection}, model_name=model_name)
    37	
    38	    return FunctionModel(fn, model_name=model_name)
    39	
    40	
    41	def failing_model(exc: Exception, *, model_name: str = "fake-down") -> FunctionModel:
    42	    def fn(messages: list[ModelMessage], info: AgentInfo) -> ModelResponse:
    43	        raise exc
    44	
    45	    return FunctionModel(fn, model_name=model_name)
    46	
    47	
    48	def text_model(text: str, *, finish_reason: str | None = None, model_name: str = "fake-text") -> FunctionModel:
    49	    def fn(messages: list[ModelMessage], info: AgentInfo) -> ModelResponse:
    50	        return ModelResponse(parts=[TextPart(text)], usage=USAGE, model_name=model_name, finish_reason=finish_reason)
    51	
    52	    return FunctionModel(fn, model_name=model_name)
    53	
    54	
    55	def last_tool_return(messages: list[ModelMessage]) -> ToolReturnPart | None:
    56	    for m in reversed(messages):
    57	        for p in getattr(m, "parts", []):
    58	            if isinstance(p, ToolReturnPart):
    59	                return p
    60	    return None
    61	
    62	
    63	Fn = Callable[[list[ModelMessage], AgentInfo], ModelResponse]

     1	"""Every numbered acceptance criterion in agent-os/specs/*/spec.md has a test named test_criterion_<n>_<slug>.
     2	
     3	A spec may opt out while in progress by listing its folder name in UNIMPLEMENTED; remove it when its tests land.
     4	"""
     5	
     6	from __future__ import annotations
     7	
     8	import re
     9	from pathlib import Path
    10	
    11	from app.config import ROOT
    12	
    13	SPECS = ROOT / "agent-os" / "specs"
    14	TESTS = Path(__file__).parent
    15	UNIMPLEMENTED: set[str] = set()
    16	
    17	CRITERION = re.compile(r"^\s*(\d+)\.\s+\S", re.MULTILINE)
    18	
    19	
    20	def criteria_of(spec: Path) -> list[int]:
    21	    text = spec.read_text(encoding="utf-8")
    22	    m = re.search(r"^##\s+Acceptance criteria\s*$(.*?)(?=^##\s|\Z)", text, re.MULTILINE | re.DOTALL | re.IGNORECASE)
    23	    return [int(n) for n in CRITERION.findall(m.group(1))] if m else []
    24	
    25	
    26	def test_every_spec_criterion_has_a_named_test():
    27	    tests_text = "\n".join(p.read_text(encoding="utf-8") for p in TESTS.glob("test_*.py"))
    28	    missing: list[str] = []
    29	    for spec in sorted(SPECS.glob("*/spec.md")):
    30	        if spec.parent.name in UNIMPLEMENTED:
    31	            continue
    32	        for n in criteria_of(spec):
    33	            if not re.search(rf"def test_criterion_{n}_\w+", tests_text):
    34	                missing.append(f"{spec.parent.name}: criterion {n}")
    35	    assert not missing, "criteria without a test:\n" + "\n".join(missing)

    76	    async def ask(self, conversation: Conversation, message: str) -> Turn:
    77	        started = time.perf_counter()
    78	
    79	        def elapsed() -> int:
    80	            return int((time.perf_counter() - started) * 1000)
    81	
    82	        if rejection := check_input(message, self.limits.max_message_chars):
    83	            if rejection.reason == "injection":
    84	                return Turn(status="refused_injection", answer=prompts.REFUSAL_INJECTION, guardrail="regex", latency_ms=elapsed())
    85	            return Turn(status="rejected_input", answer=rejection.detail, guardrail="regex", latency_ms=elapsed())
    86	
    87	        try:
    88	            rid = self.budget.reserve(self.limits.per_turn_estimate_usd, kind="turn", model="pending")
    89	        except BudgetExceeded as e:
    90	            log.warning("budget: %s", e)
    91	            return Turn(status="budget_exhausted", answer=prompts.BUDGET_EXHAUSTED, latency_ms=elapsed())
    92	
    93	        cost = 0.0
    94	        try:
    95	            verdict = await self.guard.screen(message)
    96	            if verdict.label != "in_scope":
    97	                self.budget.release(rid, failed=False)
    98	                status = "refused_injection" if verdict.label == "injection" else "refused_off_topic"
    99	                text = prompts.REFUSAL_INJECTION if verdict.label == "injection" else prompts.REFUSAL_OFF_TOPIC
   100	                return Turn(status=status, answer=text, guardrail=verdict.decided_by, latency_ms=elapsed())
   101	
   102	            result = await self.agent.run(
   103	                message,
   104	                deps=self.deps,
   105	                message_history=list(conversation.history),
   106	                usage_limits=UsageLimits(
   107	                    request_limit=self.limits.model_requests_per_turn, cost_limit=self.limits.per_turn_usd  # type: ignore[arg-type]
   108	                ),
   109	            )
   110	        except UsageLimitExceeded as e:
   111	            self.budget.release(rid, failed=True)
   112	            log.warning("turn stopped by usage limit: %s", e)
   113	            return Turn(status="cannot_answer", answer="I had to stop before finishing this answer. Try a narrower question.", latency_ms=elapsed())
   114	        except (ModelAPIError, FallbackExceptionGroup) as e:
   115	            self.budget.release(rid, failed=True)
   116	            log.error("all models failed: %r", e)
   117	            return Turn(status="unavailable", answer=prompts.UNAVAILABLE, latency_ms=elapsed())
   118	        except BaseException:
   119	            self.budget.release(rid, failed=True)
   120	            raise
   121	
   122	        usage = result.usage
   123	        model_name = served_by(result.response)
   124	        cost = float(usage.cost) if getattr(usage, "cost", None) is not None else estimate_cost(
   125	            self.prices, model_name, usage.input_tokens, usage.output_tokens
   126	        )
   127	        self.budget.settle(
   128	            rid, cost_usd=cost, model=model_name, input_tokens=usage.input_tokens,
   129	            output_tokens=usage.output_tokens, latency_ms=elapsed(),
   130	        )
   131	        new_messages = result.new_messages()
   132	        conversation.extend(new_messages)
   133	        out = result.output
   134	        return Turn(
   135	            status=out.status,
   136	            answer=out.answer,
   137	            sources=out.sources,
   138	            limitations=out.limitations,
   139	            served_by=model_name,
   140	            failed_attempts=failed_attempts(result.response),
   141	            tools_used=tools_used(new_messages),
   142	            model_requests=usage.requests,
   143	            input_tokens=usage.input_tokens,
   144	            output_tokens=usage.output_tokens,
   145	            cost_usd=cost,
   146	            latency_ms=elapsed(),
   147	        )

     1	"""Sliding-window rate limits: per IP per hour (fairness) and global per day (spend ceiling)."""
     2	
     3	from __future__ import annotations
     4	
     5	import threading
     6	import time
     7	from collections import deque
     8	from collections.abc import Callable
     9	from dataclasses import dataclass
    10	
    11	
    12	@dataclass(frozen=True)
    13	class Limited:
    14	    scope: str  # "ip" | "global"
    15	    retry_after: int
    16	    message: str
    17	
    18	
    19	class RateLimiter:
    20	    def __init__(self, *, per_ip_per_hour: int, global_per_day: int, now: Callable[[], float] = time.monotonic) -> None:
    21	        self.per_ip, self.per_day, self.now = per_ip_per_hour, global_per_day, now
    22	        self._ip: dict[str, deque[float]] = {}
    23	        self._global: deque[float] = deque()
    24	        self._lock = threading.Lock()
    25	
    26	    @staticmethod
    27	    def _prune(q: deque[float], cutoff: float) -> None:
    28	        while q and q[0] <= cutoff:
    29	            q.popleft()
    30	
    31	    def check(self, ip: str) -> Limited | None:
    32	        """Record one request for `ip`; None when allowed, else why and when to retry."""
    33	        with self._lock:
    34	            t = self.now()
    35	            self._prune(self._global, t - 86_400)
    36	            if len(self._global) >= self.per_day:
    37	                return Limited("global", int(self._global[0] + 86_400 - t) + 1, "Daily request limit reached. Try again tomorrow.")
    38	            q = self._ip.setdefault(ip, deque())
    39	            self._prune(q, t - 3_600)
    40	            if len(q) >= self.per_ip:
    41	                return Limited("ip", int(q[0] + 3_600 - t) + 1, "Too many requests from your address. Try again later.")
    42	            q.append(t)
    43	            self._global.append(t)
    44	            return None

     2	
     3	from __future__ import annotations
     4	
     5	import asyncio
     6	import secrets
     7	import time
     8	from collections import OrderedDict
     9	from collections.abc import Callable
    10	from dataclasses import dataclass, field
    11	
    12	from app.agent.core import Conversation

    14	
    15	@dataclass
    16	class Session:
    17	    conversation: Conversation
    18	    last_used: float
    19	    lock: asyncio.Lock = field(default_factory=asyncio.Lock)  # one turn at a time per session
    20	
    21	
    22	class SessionStore:
    23	    def __init__(
    24	        self, *, max_turns: int, ttl_s: float, max_sessions: int = 500, now: Callable[[], float] = time.monotonic
    25	    ) -> None:
    26	        self._max_turns, self._ttl_s, self._max_sessions, self._now = max_turns, ttl_s, max_sessions, now
    27	        self._sessions: OrderedDict[str, Session] = OrderedDict()
    28	
    29	    def get_or_create(self, session_id: str | None) -> tuple[str, Session, bool]:
    30	        """The session for `session_id` or a new one; the flag says a given id had expired."""
    31	        now = self._now()
    32	        self._sweep(now)
    33	        if session_id and (s := self._sessions.get(session_id)):
    34	            s.last_used = now
    35	            self._sessions.move_to_end(session_id)
    36	            return session_id, s, False
    37	        new_id = secrets.token_urlsafe(16)
    38	        s = Session(Conversation(max_turns=self._max_turns), last_used=now)
    39	        self._sessions[new_id] = s
    40	        while len(self._sessions) > self._max_sessions:
    41	            self._sessions.popitem(last=False)
    42	        return new_id, s, session_id is not None
    43	
    44	    def reset(self, session_id: str) -> bool:
    45	        if s := self._sessions.get(session_id):
    46	            s.conversation.reset()
    47	            return True
    48	        return False
    49	
    50	    def _sweep(self, now: float) -> None:
    51	        expired = [k for k, s in self._sessions.items() if now - s.last_used > self._ttl_s]
    52	        for k in expired:
    53	            del self._sessions[k]
    54	
    55	    def __len__(self) -> int:
    56	        return len(self._sessions)

    65	    @computed_field

    90	def check(case: EvalCase, turn: Turn) -> list[str]:
    91	    reasons: list[str] = []
    92	    if turn.status not in case.expect:
    93	        reasons.append(f"status {turn.status}, expected one of {case.expect}")
    94	    low = turn.answer.lower()
    95	    reasons += [f"answer lacks {s!r}" for s in case.must_contain if s.lower() not in low]
    96	    reasons += [f"answer contains {s!r}" for s in case.must_not_contain if s.lower() in low]
    97	    reasons += [f"tool {t!r} not called" for t in case.tools if t not in turn.tools_used]
    98	    return reasons

   101	async def run_case(service: Service, case: EvalCase, max_turns: int) -> RunResult:
   102	    conv = Conversation(max_turns=max_turns)
   103	    cost = 0.0
   104	    for prior in case.prior_turns:
   105	        cost += (await service.ask(conv, prior)).cost_usd
   106	    turn = await service.ask(conv, case.question)
   107	    reasons = check(case, turn)
   108	    return RunResult(
   109	        status=turn.status, passed=not reasons, reasons=reasons, served_by=turn.served_by, guardrail=turn.guardrail,
   110	        failed_attempts=turn.failed_attempts, tools=turn.tools_used, cost_usd=cost + turn.cost_usd,
   111	        latency_ms=turn.latency_ms, answer=turn.answer,
   112	    )
